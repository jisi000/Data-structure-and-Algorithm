dr=[2,3,4,1,0]
db=[3,0,1,2,4]
d=['dragon','ninja','iceman','lion','wolf']
n=int(input())
for i in range(n):
    print(f'Case:{i+1}')
    M=int(input())
    br,bb=M,M
    bleed=list(map(int,input().split()))
    mi=min(bleed)
    idx=0
    lr,lb=0,0
    hr={i:0 for i in range(5)}
    hb={i:0 for i in range(5)}
    while br!=-1 or bb!=-1:
        rer,reb=bleed[dr[lr%5]],bleed[db[lb%5]]
        if br!=-1:
            if br < mi:
                print(f'{idx:03d} red headquarter stops making warriors')
                br=-1
            else:
                while br<rer:
                    lr+=1
                    rer=bleed[dr[lr%5]]
                br-=rer
                hr[dr[lr%5]]+=1
                print(f'{idx:03d} red {d[dr[lr%5]]} {idx+1} born with strength {rer},{hr[dr[lr%5]]} {d[dr[lr%5]]} in red headquarter')
                lr+=1
        if bb!=-1:
            if bb<mi:
                print(f'{idx:03d} blue headquarter stops making warriors')
                bb=-1
            else:
                while bb<reb:
                    lb+=1
                    reb=bleed[db[lb%5]]
                bb-=reb
                hb[db[lb%5]]+=1
                print(f'{idx:03d} blue {d[db[lb%5]]} {idx+1} born with strength {reb},{hb[db[lb%5]]} {d[db[lb%5]]} in blue headquarter')
                lb+=1
        idx+=1
