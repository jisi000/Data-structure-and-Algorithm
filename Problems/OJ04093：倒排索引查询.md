利用长度固定为N的列表代表文档，‘1’代表这个词存在于此文档，‘-1’代表不存在

列表及其对应编号用字典存储，定义compare函数比较查询和文档是否相符

```python
N=int(input())
d={}
def compare(s1,s2):
    for i in range(N):
        if s2[i]!= '0' and s1[i]!=s2[i]:
            return False
    return True
for i in range(N):
    data=list(map(int,input().split()))
    s_data=set(data[1:])
    for doct in s_data:
        if doct not in d:
            d[doct]=['-1']*N
        d[doct][i]='1'
q=sorted(d.keys())
M=int(input())
for _ in range(M):
    inquire=list(input().split())
    ans=[]
    for j in q:
        if compare(d[j],inquire):
            ans.append(j)
    if ans:
        print(*ans)
    else:
        print('NOT FOUND')
```

第一次知道集合还有交集（&）并集（|）补集（-）这样方便的用法，比第一次做对所有文件一个个查询快了三倍。

```python
import sys
data=list(map(int,sys.stdin.read().split()))
index=0
N=data[index]
index+=1
d=[]
for _ in range(N):
    l=data[index]
    index+=1
    d.append(data[index:index+l])
    index+=l
M=data[index]
index+=1
for _ in range(M):
    query=data[index:index+N]
    index+=N
    in_set=[]
    ex_set=set()
    for i in range(N):
        if query[i]==1:
            in_set.append(set(d[i]))
        elif query[i]==-1:
            ex_set.update(d[i])
    ans=set()
    if in_set:
        ans=in_set[0]
        for ins in in_set[1:]:
            ans=ans&ins
        ans=ans-ex_set
    if ans:
        print(*sorted(list(ans)))
    else:
        print('NOT FOUND')
```
