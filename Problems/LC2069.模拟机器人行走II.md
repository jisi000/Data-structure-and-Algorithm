实际上机器人只在边框上移动，因此记录边框上每个位置的方向和坐标，用索引查找即可。需要注意[0,0]只有一开始是朝东的，移动后就会朝南。

```python
class Robot:

    def __init__(self, width: int, height: int):
        self.pos=list()
        self.dir=list()
        self.idx=0
        self.move=False
        for i in range(width):
            self.pos.append([i,0])
            self.dir.append(0)
        for j in range(1,height):
            self.pos.append([width-1,j])
            self.dir.append(1)
        for i in range(width-2,-1,-1):
            self.pos.append([i,height-1])
            self.dir.append(2)
        for j in range(height-2,0,-1):
            self.pos.append([0,j])
            self.dir.append(3)
        self.dir[0]=3
        self.l=len(self.pos)
    def step(self, num: int) -> None:
        self.move=True
        self.idx+=num

    def getPos(self) -> List[int]:
        return self.pos[self.idx%self.l]

    def getDir(self) -> str:
        to_dir=['East','North','West','South']
        if not self.move:
            return 'East'
        return to_dir[self.dir[self.idx%self.l]]
```
