遍历，每找到一个每访问过的位置就使用一次dfs。在dfs中，遍历四个方向，如果没有墙，就走到这个位置。如果没有墙的位置离开了城堡范围，说明这个房间是开放的，不是合法房间。判断墙时使用位运算。

```python
m=int(input())
n=int(input())
grid=[list(map(int,input().split()))for _ in range(m)]
visited=[[False]*n for _ in range(m)]
block_num=[0]
mx_cell=[0]
cell_num=[0]
def dfs(i,j):
    d={1:(0,-1),2:(-1,0),4:(0,1),8:(1,0)}
    block_num[0]+=1
    visited[i][j]=True
    stack=[(i,j)]
    re=1
    flag=True
    while stack:
        x,y=stack.pop()
        mask=grid[x][y]
        for num in [1,2,4,8]:
            if not (num&mask):
                nx,ny=x+d[num][0],y+d[num][1]
                if 0<=nx<m and 0<=ny<n:
                    if not visited[nx][ny]:
                        block_num[0]+=1
                        visited[nx][ny]=True
                        stack.append((nx,ny))
                        re+=1
                else:
                    flag=False
    if flag:
        cell_num[0]+=1
        mx_cell[0]=max(mx_cell[0],re)
for i in range(m):
    for j in range(n):
        if not visited[i][j]:
            dfs(i,j)
print(cell_num[0])
print(mx_cell[0])
```
