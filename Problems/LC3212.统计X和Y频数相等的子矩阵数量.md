因为子矩阵都包含[0] [0]，因此不需要使用标准前缀和，只需要用简略版依次遍历。

```python
class Solution:
    def numberOfSubmatrices(self, grid: List[List[str]]) -> int:
        r,c=len(grid),len(grid[0])        
        total_x=[0]*c
        total_y=[0]*c
        count=0
        for i in range(r):
            x,y=0,0
            for j in range(c):
                if grid[i][j]=='X':
                    x+=1
                elif grid[i][j]=='Y':
                    y+=1
                total_x[j]+=x
                total_y[j]+=y
                if total_x[j]>=1 and total_x[j]==total_y[j]:
                    count+=1
        return count
```
