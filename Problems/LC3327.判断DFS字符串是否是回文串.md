后序遍历，记录该位置在整体遍历结果中的区间范围，然后对整体遍历结果跑一次马拉车，判断每个区间是否是回文串。注意索引换算。

```python
class Solution:
    def findAnswer(self, parent: List[int], s: str) -> List[bool]:
        n=len(s)
        interval=[[-1,-1] for _ in range(n)]
        children=[[] for _ in range(n)]
        for i,p in enumerate(parent):
            if p!=-1:
                children[p].append(i)
        n_s=['']*n
        loc=[0]
        def dfs(m): #后序遍历
            interval[m][0]=loc[0]
            re=i
            for child in children[m]:
                dfs(child)
            n_s[loc[0]]=s[m]
            loc[0]+=1
            interval[m][1]=loc[0]
        dfs(0)
        pal='#'.join(['$']+n_s+['&']) #马拉车
        halfl=[0]*(len(pal)-2)
        halfl[1]=1
        c,r=0,0
        for i in range(2,len(pal)-2):
            l=1
            if i<=r:
                l=min(halfl[(2*c)-i],r-i+1)
            while pal[i-l]==pal[i+l]:
                l+=1
                c=i
                r=i+l-1
            halfl[i]=l
        def judge(a,b):
            return halfl[a+b+1]>=(b-a)
        ans=[judge(x,y) for x,y in interval]
        return ans
```
